```python
from abc import ABC, abstractmethod
from typing import Dict, List, Tuple, Optional

class ContextAdjuster(ABC):
    """上下文调整器基类"""
    
    @abstractmethod
    def adjust(self, scores: List[float], context: Dict, actions: List) -> List[float]:
        """
        根据上下文调整评分
        
        Args:
            scores: 原始评分列表
            context: 上下文信息
            actions: 动作列表
            
        Returns:
            调整后的评分列表
        """
        pass
    
    @abstractmethod
    def get_name(self) -> str:
        """获取调整器名称"""
        pass
```

```python
class NextPlayerAdjuster(ContextAdjuster):
    """下家牌数调整器"""
    
    def adjust(self, scores: List[float], context: Dict, actions: List) -> List[float]:
        next_player_remain = context.get("next_player_remain", 27)
        adjusted_scores = scores.copy()
        
        if next_player_remain == 1:
            # 下家只剩1张，降低单张优先级
            for i, action in enumerate(actions):
                if len(action) > 0 and action[0] == "Single":
                    adjusted_scores[i] *= 0.5  # 降低单张优先级
        elif next_player_remain <= 3:
            # 下家牌数很少，提高压制性动作优先级
            for i, action in enumerate(actions):
                if len(action) > 0 and action[0] in ["Bomb", "Trips"]:
                    adjusted_scores[i] *= 1.3  # 提高压制性动作优先级
        
        return adjusted_scores
    
    def get_name(self) -> str:
        return "NextPlayerAdjuster"


class PassCountAdjuster(ContextAdjuster):
    """PASS次数调整器"""
    
    def adjust(self, scores: List[float], context: Dict, actions: List) -> List[float]:
        pass_count = context.get("pass_count", 0)
        adjusted_scores = scores.copy()
        
        if pass_count >= 5:
            # PASS次数过多，提高出牌优先级
            for i, action in enumerate(actions):
                if len(action) > 0 and action[0] != "PASS":
                    adjusted_scores[i] *= 1.2  # 提高出牌优先级
        elif pass_count >= 3:
            # 适度提高出牌优先级
            for i, action in enumerate(actions):
                if len(action) > 0 and action[0] != "PASS":
                    adjusted_scores[i] *= 1.1
        
        return adjusted_scores
    
    def get_name(self) -> str:
        return "PassCountAdjuster"


class EndgameAdjuster(ContextAdjuster):
    """残局调整器"""
    
    def adjust(self, scores: List[float], context: Dict, actions: List) -> List[float]:
        is_endgame = context.get("is_endgame", False)
        endgame_type = context.get("endgame_type", "normal")
        adjusted_scores = scores.copy()
        
        if is_endgame:
            if endgame_type == "rush":
                # 冲刺型残局，提高大牌型优先级
                for i, action in enumerate(actions):
                    if len(action) > 0 and action[0] in ["Bomb", "Straight", "ThreeWithTwo"]:
                        adjusted_scores[i] *= 1.5
            elif endgame_type == "defend":
                # 防守型残局，降低出牌优先级
                for i, action in enumerate(actions):
                    if len(action) > 0 and action[0] == "PASS":
                        adjusted_scores[i] *= 1.8  # 提高PASS优先级
                    else:
                        adjusted_scores[i] *= 0.7  # 降低出牌优先级
        
        return adjusted_scores
    
    def get_name(self) -> str:
        return "EndgameAdjuster"


class TeammateAdjuster(ContextAdjuster):
    """队友状态调整器"""
    
    def adjust(self, scores: List[float], context: Dict, actions: List) -> List[float]:
        teammate_cards = context.get("teammate_cards", 27)
        teammate_is_greater = context.get("teammate_is_greater", False)
        adjusted_scores = scores.copy()
        
        if teammate_is_greater and teammate_cards <= 8:
            # 队友领先且牌数少，提高PASS优先级
            for i, action in enumerate(actions):
                if len(action) > 0 and action[0] == "PASS":
                    adjusted_scores[i] *= 2.0  # 大幅提高PASS优先级
                else:
                    adjusted_scores[i] *= 0.3  # 大幅降低出牌优先级
        
        return adjusted_scores
    
    def get_name(self) -> str:
        return "TeammateAdjuster"
```

```python
class DynamicPrioritySystem:
    """动态优先级系统"""
    
    def __init__(self, config: Dict = None):
        self.config = config or {}
        self.context_adjusters = [
            TeammateAdjuster(),      # 最高优先级
            EndgameAdjuster(),
            NextPlayerAdjuster(),
            PassCountAdjuster(),
        ]
        
        # 可配置的调整器权重
        self.adjuster_weights = self.config.get("adjuster_weights", {
            "TeammateAdjuster": 1.0,
            "EndgameAdjuster": 0.9,
            "NextPlayerAdjuster": 0.8,
            "PassCountAdjuster": 0.7,
        })
    
    def evaluate_with_context(self, action_list: List, base_scores: List[float], 
                            message: Dict) -> List[Tuple[int, float]]:
        """
        根据上下文动态调整优先级
        
        Args:
            action_list: 动作列表
            base_scores: 基础评分
            message: 游戏状态消息
            
        Returns:
            [(action_index, adjusted_score), ...] 按分数降序排列
        """
        # 构建上下文
        context = self._build_context(message)
        
        # 应用所有调整器
        adjusted_scores = base_scores.copy()
        adjustment_log = {}
        
        for adjuster in self.context_adjusters:
            adjuster_name = adjuster.get_name()
            weight = self.adjuster_weights.get(adjuster_name, 1.0)
            
            # 获取调整后的分数
            temp_scores = adjuster.adjust(adjusted_scores, context, action_list)
            
            # 应用权重
            for i in range(len(adjusted_scores)):
                adjustment = temp_scores[i] - adjusted_scores[i]
                adjusted_scores[i] += adjustment * weight
            
            # 记录调整日志
            adjustment_log[adjuster_name] = {
                "weight": weight,
                "applied": True
            }
        
        # 排序并返回
        indexed_scores = [(i, score) for i, score in enumerate(adjusted_scores)]
        indexed_scores.sort(key=lambda x: x[1], reverse=True)
        
        # 记录调整信息
        self._log_adjustment(context, adjustment_log, indexed_scores[:3])
        
        return indexed_scores
    
    def _build_context(self, message: Dict) -> Dict:
        """构建上下文信息"""
        my_pos = message.get("myPos", 0)
        teammate_pos = (my_pos + 2) % 4
        public_info = message.get("publicInfo", [])
        greater_pos = message.get("greaterPos", -1)
        
        # 获取剩余牌数
        my_cards = len(message.get("handCards", []))
        teammate_cards = 27
        next_player_cards = 27
        
        if len(public_info) > teammate_pos and isinstance(public_info[teammate_pos], dict):
            teammate_cards = public_info[teammate_pos].get('rest', 27)
        
        next_pos = (my_pos + 1) % 4
        if len(public_info) > next_pos and isinstance(public_info[next_pos], dict):
            next_player_cards = public_info[next_pos].get('rest', 27)
        
        # 判断残局
        from .endgame_strategy import _is_endgame_enhanced
        is_endgame, endgame_type = _is_endgame_enhanced(message)
        
        return {
            "my_cards": my_cards,
            "teammate_cards": teammate_cards,
            "next_player_remain": next_player_cards,
            "teammate_is_greater": greater_pos == teammate_pos,
            "is_endgame": is_endgame,
            "endgame_type": endgame_type,
            "pass_count": self._get_pass_count(message),
        }
    
    def _get_pass_count(self, message: Dict) -> int:
```
