```python
from abc import ABC, abstractmethod
from typing import Dict, List, Optional

class TeammateProtectionRule(ABC):
    """队友保护规则基类"""
    
    @abstractmethod
    def evaluate(self, message: dict, context: dict) -> float:
        """
        评估保护需求
        
        Args:
            message: 游戏状态消息
            context: 上下文信息
            
        Returns:
            保护需求分数 (0.0-1.0)
        """
        pass
    
    @abstractmethod
    def get_name(self) -> str:
        """获取规则名称"""
        pass
```

```python
class HighValueProtectionRule(TeammateProtectionRule):
    """高牌值保护规则（学习lalala）"""
    
    def evaluate(self, message: dict, context: dict) -> float:
        cur_action = message.get("curAction")
        greater_pos = message.get("greaterPos", -1)
        teammate_pos = context.get("teammate_pos", -1)
        
        # 如果队友是最大动作者
        if greater_pos == teammate_pos and cur_action:
            cur_val = self._get_card_value(cur_action)
            
            if cur_val >= 15:  # 2或王
                return 1.0  # 高保护需求
            elif cur_val >= 14:  # A
                return 0.7  # 中等保护需求
            elif cur_val >= 12:  # Q或更大
                return 0.4  # 低保护需求
        
        return 0.0
    
    def get_name(self) -> str:
        return "HighValueProtection"
    
    def _get_card_value(self, action) -> int:
        """获取牌值"""
        if not action or len(action) < 2:
            return 0
        
        rank = action[1]
        rank_map = {
            '3': 3, '4': 4, '5': 5, '6': 6, '7': 7, '8': 8, '9': 9,
            'T': 10, '10': 10, 'J': 11, 'Q': 12, 'K': 13, 'A': 14,
            '2': 15, 'B': 16, 'R': 17
        }
        return rank_map.get(str(rank).upper(), 0)


class LowCardCountProtectionRule(TeammateProtectionRule):
    """低牌数保护规则"""
    
    def evaluate(self, message: dict, context: dict) -> float:
        teammate_cards = context.get("teammate_cards", 27)
        
        if teammate_cards <= 1:
            return 1.0  # 绝对保护
        elif teammate_cards <= 2:
            return 0.9  # 极高保护
        elif teammate_cards <= 5:
            return 0.7  # 高度保护
        elif teammate_cards <= 8:
            return 0.4  # 适度保护
        
        return 0.0
    
    def get_name(self) -> str:
        return "LowCardCountProtection"


class CriticalStageProtectionRule(TeammateProtectionRule):
    """关键阶段保护规则"""
    
    def evaluate(self, message: dict, context: dict) -> float:
        # 检查是否是关键阶段
        my_cards = len(message.get("handCards", []))
        teammate_cards = context.get("teammate_cards", 27)
        opponents_min = min(context.get("opponents_cards", [27, 27]))
        
        # 如果对手也很少牌，需要更积极保护队友
        if opponents_min <= 5 and teammate_cards <= 8:
            return 0.8
        elif opponents_min <= 3 and teammate_cards <= 10:
            return 0.6
        
        return 0.0
    
    def get_name(self) -> str:
        return "CriticalStageProtection"


class BombProtectionRule(TeammateProtectionRule):
    """炸弹保护规则"""
    
    def evaluate(self, message: dict, context: dict) -> float:
        cur_action = message.get("curAction")
        greater_pos = message.get("greaterPos", -1)
        teammate_pos = context.get("teammate_pos", -1)
        
        # 如果队友打了炸弹，绝对不能压制
        if (greater_pos == teammate_pos and cur_action and 
            len(cur_action) > 0 and cur_action[0] == "Bomb"):
            return 1.0  # 绝对保护
        
        return 0.0
    
    def get_name(self) -> str:
        return "BombProtection"
```

### 步骤5：创建保护策略管理器
**在同一文件中添加管理器类**：

```python
class TeammateProtectionStrategy:
    """队友保护策略管理器"""
    
    def __init__(self, config: dict = None):
        self.config = config or {}
        self.protection_rules = [
            BombProtectionRule(),           # 最高优先级
            HighValueProtectionRule(),
            LowCardCountProtectionRule(),
            CriticalStageProtectionRule(),
        ]
        
        # 可配置的权重
        self.rule_weights = self.config.get("protection_weights", {
            "BombProtection": 1.0,
            "HighValueProtection": 0.8,
            "LowCardCountProtection": 0.9,
            "CriticalStageProtection": 0.7,
        })
    
    def should_protect(self, message: dict) -> tuple[bool, dict]:
        """
        判断是否应该保护队友
        
        Returns:
            (should_protect, protection_info)
        """
        context = self._build_context(message)
        protection_score = 0.0
        rule_scores = {}
        
        # 综合所有保护规则
        for rule in self.protection_rules:
            rule_name = rule.get_name()
            score = rule.evaluate(message, context)
            weight = self.rule_weights.get(rule_name, 1.0)
            weighted_score = score * weight
            
            protection_score += weighted_score
            rule_scores[rule_name] = {
                "raw_score": score,
                "weight": weight,
                "weighted_score": weighted_score
            }
        
        # 动态阈值
        threshold = self._get_dynamic_threshold(message, context)
        should_protect = protection_score >= threshold
        
        protection_info = {
            "total_score": protection_score,
            "threshold": threshold,
            "rule_scores": rule_scores,
            "context": context
        }
        
        return should_protect, protection_info
    
    def _build_context(self, message: dict) -> dict:
        """构建上下文信息"""
        my_pos = message.get("myPos", 0)
        teammate_pos = (my_pos + 2) % 4
        public_info = message.get("publicInfo", [])
        
        # 获取剩余牌数
        teammate_cards = 27
        opponents_cards = [27, 27]
        
        if len(public_info) > teammate_pos and isinstance(public_info[teammate_pos], dict):
            teammate_cards = public_info[teammate_pos].get('rest', 27)
        
        opponent_positions = [(my_pos + 1) % 4, (my_pos + 3) % 4]
        for i, pos in enumerate(opponent_positions):
            if len(public_info) > pos and isinstance(public_info[pos], dict):
                opponents_cards[i] = public_info[pos].get('rest', 27)
        
        return {
            "teammate_pos": teammate_pos,
            "teammate_cards": teammate_cards,
            "opponents_cards": opponents_cards,
            "my_cards": len(message.get("handCards", [])),
        }
    
    def _get_dynamic_threshold(self, message: dict, context: dict) -> float:
        """获取动态保护阈值"""
        base_threshold = 0.5
        
        # 根据游戏阶段调整
        my_cards = context.get("my_cards", 27)
        teammate_cards = context.get("teammate_cards", 27)
        
        # 残局阶段，降低阈值（更容易保护）
        if my_cards <= 10 or teammate_cards <= 10:
            base_threshold *= 0.8
        
        # 如果对手牌数很少，提高阈值（更谨慎保护）
        min_opponent = min(context.get("opponents_cards", [27, 27]))
        if min_opponent <= 5:
            base_threshold *= 1.2
        
        return max(0.1, min(1.0, base_threshold))
```
