```python
class DoudizhuRLEnv(gym.Env):
    def __init__(self, max_rounds=100):
        super().__init__()

        # 状态空间：游戏状态的向量化表示
        self.observation_space = spaces.Box(
            low=0, high=1,
            shape=(512,),  # 与现有网络兼容
            dtype=np.float32
        )

        # 动作空间：所有可能的出牌动作
        self.action_space = spaces.Discrete(10000)  # 预估动作数量

        # 奖励函数参数
        self.win_reward = 1.0
        self.lose_reward = -1.0
        self.step_penalty = -0.01  # 鼓励快速决策

        # 游戏状态
        self.game_state = None
        self.current_player = 0
        self.round_count = 0
        self.max_rounds = max_rounds

    def reset(self):
        """重置环境到初始状态"""
        self.game_state = self._initialize_game()
        self.current_player = 0
        self.round_count = 0

        observation = self._get_observation()
        return observation

    def step(self, action):
        """执行动作并返回下一个状态、奖励、是否结束"""
        # 执行动作
        valid_action = self._decode_action(action)
        success = self._execute_action(valid_action)

        if not success:
            # 无效动作给予惩罚
            reward = -0.5
            done = False
        else:
            # 计算奖励
            reward = self._calculate_reward()

            # 检查游戏结束
            done = self._is_game_end()

            if done:
                # 最终奖励基于胜负
                if self._is_current_player_winner():
                    reward += self.win_reward
                else:
                    reward += self.lose_reward

        # 获取下一个观察
        observation = self._get_observation()

        # 切换到下一个玩家
        self.current_player = (self.current_player + 1) % 4
        self.round_count += 1

        # 检查回合限制
        if self.round_count >= self.max_rounds:
            done = True

        return observation, reward, done, {}

    def _calculate_reward(self):
        """计算步骤奖励"""
        reward = self.step_penalty  # 基础惩罚

        # 策略有效性奖励
        strategy_score = self._evaluate_strategy_effectiveness()
        reward += strategy_score * 0.1

        # 局面改善奖励
        position_improvement = self._calculate_position_improvement()
        reward += position_improvement * 0.05

        return reward
```

### PPO算法实现
```python
class DoudizhuPPO:
    def __init__(self, policy_net, value_net, lr=3e-4):
        self.policy_net = policy_net
        self.value_net = value_net

        self.optimizer = torch.optim.Adam([
            {'params': policy_net.parameters()},
            {'params': value_net.parameters()}
        ], lr=lr)

        # PPO参数
        self.clip_epsilon = 0.2
        self.value_coef = 0.5
        self.entropy_coef = 0.01
        self.max_grad_norm = 0.5

        # 经验缓冲区
        self.memory = PPOMemory()

    def update(self, batch_size=64):
        """执行PPO更新"""
        # 从经验缓冲区采样
        states, actions, old_log_probs, rewards, dones, values = \
            self.memory.sample(batch_size)

        # 计算优势函数
        advantages = self._calculate_advantages(rewards, values, dones)

        # 多轮更新
        for _ in range(10):  # PPO epochs
            # 计算策略损失
            new_log_probs = self.policy_net.get_log_probs(states, actions)
            ratio = torch.exp(new_log_probs - old_log_probs)

            policy_loss1 = advantages * ratio
            policy_loss2 = advantages * torch.clamp(ratio, 1-self.clip_epsilon, 1+self.clip_epsilon)
            policy_loss = -torch.min(policy_loss1, policy_loss2).mean()

            # 计算价值损失
            new_values = self.value_net(states)
            value_loss = F.mse_loss(new_values, rewards)

            # 计算熵奖励
            entropy = self.policy_net.get_entropy(states)
            entropy_loss = -entropy.mean()

            # 总损失
            total_loss = (policy_loss +
                         self.value_coef * value_loss +
                         self.entropy_coef * entropy_loss)

            # 梯度更新
            self.optimizer.zero_grad()
            total_loss.backward()
            torch.nn.utils.clip_grad_norm_(self.policy_net.parameters(), self.max_grad_norm)
            torch.nn.utils.clip_grad_norm_(self.value_net.parameters(), self.max_grad_norm)
            self.optimizer.step()

        return {
            'policy_loss': policy_loss.item(),
            'value_loss': value_loss.item(),
            'entropy': entropy.mean().item()
        }
```

### 自对弈训练框架
```python
class SelfPlayTrainer:
    def __init__(self, agent_class, num_agents=4, pool_size=10):
        self.agent_class = agent_class
        self.num_agents = num_agents

        # 对手池：存储历史版本
        self.opponent_pool = deque(maxlen=pool_size)
        self.current_agents = []

        # 训练统计
        self.training_stats = {
            'wins': [],
            'losses': [],
            'draws': [],
            'strategy_diversity': []
        }

    def initialize_pool(self):
        """初始化对手池"""
        for _ in range(self.pool_size):
            agent = self.agent_class()
            self.opponent_pool.append(agent)

    def train_epoch(self, num_games=100):
        """执行一个训练周期"""
        epoch_stats = {'wins': 0, 'losses': 0, 'draws': 0}

        for game_idx in range(num_games):
            # 从池中选择对手
            opponents = self._select_opponents()

            # 创建新的智能体进行训练
            training_agent = self.agent_class()

            # 执行对局
            game_result = self._play_game(training_agent, opponents)

            # 更新统计
            if game_result == 'win':
                epoch_stats['wins'] += 1
            elif game_result == 'loss':
                epoch_stats['losses'] += 1
            else:
                epoch_stats['draws'] += 1

            # 学习更新
            training_agent.update_policy(game_result)

        # 评估是否加入对手池
        win_rate = epoch_stats['wins'] / num_games
        if win_rate > 0.6:  # 胜率超过60%加入池中
            self.opponent_pool.append(training_agent)

        return epoch_stats

    def _select_opponents(self):
        """选择对手组合"""
        # 保证多样性：选择不同历史版本
        selected = random.sample(self.opponent_pool, self.num_agents - 1)

        # 偶尔加入最新版本进行对抗
        if random.random() < 0.3 and len(self.current_agents) > 0:
            selected[-1] = random.choice(self.current_agents)

        return selected
```
