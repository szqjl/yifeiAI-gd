## 🔧 技术实现要点

### 评估框架设计
```python
class GameOrientedEvaluator:
    def __init__(self):
        self.win_rate_weight = 0.5
        self.strategy_weight = 0.2
        self.decision_weight = 0.2
        self.prediction_weight = 0.1

    def evaluate_model(self, model, game_records):
        """计算综合评估分数"""
        win_rate = self._calculate_win_rate(game_records)
        strategy_effectiveness = self._analyze_strategy_effectiveness(game_records)
        decision_quality = self._assess_decision_quality(game_records)
        prediction_accuracy = self._measure_prediction_accuracy(game_records)

        total_score = (win_rate * self.win_rate_weight +
                      strategy_effectiveness * self.strategy_weight +
                      decision_quality * self.decision_weight +
                      prediction_accuracy * self.prediction_weight)

        return total_score
```

### 胜率导向损失函数
```python
class GameOrientedLoss(nn.Module):
    def __init__(self):
        super().__init__()
        self.action_loss = ActionPredictionLoss()
        self.strategy_loss = StrategyClassificationLoss()
        self.win_rate_loss = WinRatePredictionLoss()

        # 动态权重调整
        self.action_weight = nn.Parameter(torch.tensor(1.0))
        self.strategy_weight = nn.Parameter(torch.tensor(0.3))
        self.win_weight = nn.Parameter(torch.tensor(2.0))

    def forward(self, predictions, targets, game_results):
        action_loss = self.action_loss(predictions['actions'], targets['actions'])
        strategy_loss = self.strategy_loss(predictions['strategy'], targets['strategy'])
        win_loss = self.win_rate_loss(predictions['win_prob'], game_results['win'])

        total_loss = (self.action_weight * action_loss +
                     self.strategy_weight * strategy_loss +
                     self.win_weight * win_loss)

        return total_loss
```
