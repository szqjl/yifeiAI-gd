# -*- coding: utf-8 -*-
"""
yf1_v6 - YiFei AI V6 Client (Player 0)
基于硬编码优化方案的增强版本
"""
import asyncio
import websockets
import json
import sys
import logging
from pathlib import Path
from datetime import datetime

# Add paths
sys.path.insert(0, str(Path(__file__).parent.parent))
sys.path.insert(0, str(Path(__file__).parent.parent.parent))

from decision.decision_engine import DecisionEngine
from game_logic.enhanced_state import EnhancedGameStateManager
from communication.game_recorder import GameRecorder

# 配置日志
log_dir = Path(__file__).parent.parent.parent / "logs"
log_dir.mkdir(exist_ok=True)
log_filename = log_dir / f"yf1_v6_{datetime.now().strftime('%Y%m%d_%H%M%S')}.log"

logging.basicConfig(
    level=logging.INFO,
    format='[%(asctime)s] [%(name)s] [%(levelname)s] %(message)s',
    datefmt='%H:%M:%S',
    handlers=[
        logging.FileHandler(log_filename, encoding='utf-8'),
        logging.StreamHandler()
    ]
)

class YF1_V6_Client:
    """
    YiFei AI V6 Client - 基于硬编码优化方案
    
    主要改进：
    - 智能残局处理
    - 增强队友保护
    - 动态优先级系统
    """
    
    def __init__(self, player_id=0):
        self.player_id = player_id
        self.user_info = "yf1_v6"
        self.websocket = None
        self.logger = logging.getLogger(f"yf1_v6")
        
        # 使用增强的决策引擎
        state_manager = EnhancedGameStateManager()
        self.decision_engine = DecisionEngine(state_manager)
        
        # 统计信息
        self.decision_count = 0
        self.game_count = 0
        
        # 游戏记录器
        self.game_recorder = GameRecorder(player_id, "yf1_v6")
        
        self.logger.info(f"✓ yf1_v6 initialized (Player {player_id})")
    
    async def connect(self):
        """连接到游戏服务器"""
        uri = f"ws://127.0.0.1:23456/game/{self.user_info}"
        try:
            self.websocket = await websockets.connect(
                uri,
                ping_timeout=None,
                close_timeout=10
            )
            self.logger.info(f"✓ Connected to server: {uri}")
            await self.handle_messages()
        except Exception as e:
            self.logger.error(f"✗ Connection error: {e}")
    
    async def handle_messages(self):
        """处理服务器消息"""
        try:
            async for message in self.websocket:
                try:
                    data = json.loads(message)
                    await self.process_message(data)
                except json.JSONDecodeError as e:
                    self.logger.error(f"JSON decode error: {e}")
                except Exception as e:
                    self.logger.error(f"Message processing error: {e}")
        except websockets.exceptions.ConnectionClosed:
            self.logger.info("Connection closed")
        except Exception as e:
            self.logger.error(f"Message handling error: {e}")
    
    async def process_message(self, data):
        """处理具体消息"""
        msg_type = data.get("type")
        
        if msg_type == "act":
            await self.handle_action_request(data)
        elif msg_type == "notify":
            self.handle_notification(data)
        else:
            self.logger.debug(f"Unknown message type: {msg_type}")
    
    async def handle_action_request(self, data):
        """处理行动请求"""
        try:
            start_time = time.time()
            
            # 使用增强的决策引擎
            action_index = self.decision_engine.decide(data)
            
            decision_time = time.time() - start_time
            self.decision_count += 1
            
            # 发送响应
            response = {"actIndex": action_index}
            await self.websocket.send(json.dumps(response))
            
            # 记录决策
            self.game_recorder.record_decision(data, action_index, decision_time)
            
            self.logger.info(
                f"Decision {self.decision_count}: action={action_index}, "
                f"time={decision_time:.3f}s"
            )
            
        except Exception as e:
            self.logger.error(f"Action handling error: {e}")
            # 紧急回退
            await self.websocket.send(json.dumps({"actIndex": 0}))
    
    def handle_notification(self, data):
        """处理通知消息"""
        stage = data.get("stage", "")
        if stage == "gameOver":
            self.game_count += 1
            self.logger.info(f"Game {self.game_count} completed")
            self.game_recorder.record_game_end(data)

# 启动客户端
async def main():
    client = YF1_V6_Client(0)
    await client.connect()

if __name__ == "__main__":
    import time
    asyncio.run(main())
